The bet why it's ours, not a clone

Rive's .riv is binary; Lottie is opaque After-Effects export JSON. Neither can be reliably written by an LLM, because both were designed for a GUI editor with a human driving — the file is a byproduct. Glamour inverts that: the format is the product. A .glam file is small, fully named, and prompt-sized, so Claude composes interactive animations from a known palette instead of guessing. That one document is the AI's API, the studio's save file, and the player's input — all at once.

The .glam format this is the moat

// everything named · on = node.event or input-condition · set = node.prop
{
  "schema": "glamour/v0",
  "canvas": { "w": 640, "h": 300, "bg": "#12151b" },
  "inputs": { "progress": 0 },
  "nodes": [
    { "id": "orb", "type": "circle", "x": 200, "y": 150, "r": 48, "fill": "#4c7dff" },
    { "id": "bar", "type": "rect", "x": 300, "y": 170, "w": 20, "h": 10, "fill": "#5ad67d" }
  ],
  "bind": [ { "node": "bar", "prop": "w", "expr": "lerp(20, 220, progress)" } ],
  "machine": {
    "initial": "idle",
    "states": {
      "idle":   { "set": { "orb.r": 48 }, "on": { "orb.click": "active" } },
      "active": { "set": { "orb.r": 72 }, "on": { "orb.click": "idle" } }
    },
    "transition": { "ms": 250, "ease": "easeOut" }
  }
}
